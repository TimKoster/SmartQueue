      "command": "sbatch",
      "options": {

      ],
    },

      "label": "submit job",
      "type": "shell",

      },
      "group": "build"
    },
  ],
  "inputs": [
    {
        "id": "param1",
        "description": "Node partition:",
        "default": "A",
        "type": "promptString"
    },
    {
        "id": "param2",
        "description": "Display name:",
        "default": "Name",
        "type": "promptString"
    },
  ]
